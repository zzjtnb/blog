# 检测PC端或者移动端
```js
/**
*检测PC端或移动端
*/
if (/Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent)) {
    console.log("移动端");
} else {
     console.log("PC端");
}
```
# 检测是否为ios
```js
if (navigator.userAgent.indexOf("iPhone") > 0 || navigator.userAgent.indexOf("iPad") > 0) {
    console.log("ios")
} else {
    console.log("PC或者Android")      
}
```
